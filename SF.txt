  /** Non-standard codes some systems use. */
  const ISO_ALIASES = { UK: 'GB', EL: 'GR' };

  /** "Côte d'Ivoire" -> "cotedivoire", "St Lucia" -> "saintlucia", "The Gambia" -> "gambia". */
  function normName(s) {
    return String(s || '')
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .toLowerCase()
      .replace(/&/g, ' and ')
      .replace(/\bst\.?\s/g, 'saint ')
      .replace(/^\s*the\s+/, '')
      .replace(/[^a-z0-9]/g, '');
  }

  const COUNTRIES = COUNTRY_DATA.trim().split('\n').map((line) => {
    const [iso, name, dial, alias = ''] = line.split('|').map((s) => s.trim());
    return { iso, name, dial, aliases: alias ? alias.split(';').map((s) => s.trim()).filter(Boolean) : [] };
  });
  const BY_ISO = new Map(COUNTRIES.map((c) => [c.iso, c]));
  const BY_NAME = new Map();
  for (const c of COUNTRIES) {
    for (const n of [c.name, ...c.aliases]) {
      const k = normName(n);
      if (k && !BY_NAME.has(k)) BY_NAME.set(k, c);
    }
  }
  const DIAL_CODES = new Set(COUNTRIES.map((c) => c.dial));

  function resolveCountry(value) {
    const v = String(value || '').trim();
    if (!v) return null;
    if (/^[A-Za-z]{2}$/.test(v)) {
      const up = v.toUpperCase();
      return BY_ISO.get(ISO_ALIASES[up] || up) || null;
    }
    return BY_NAME.get(normName(v)) || null;
  }

  /** Phone country codes are prefix-free, so the longest matching prefix is the right one. */
  function dialFromPhone(phone) {
    for (let len = 3; len >= 1; len--) {
      const p = String(phone || '').slice(0, len);
      if (p.length === len && DIAL_CODES.has(p)) return p;
    }
    return null;
  }

  /* =====================================================================
   * 3. EVERGENT PARSER
   * ===================================================================== */
  const LABELS = new Set([
    'all apps / modules', 'quick links', 'account summary', 'partner', 'cp customer id', 'sp account id',
    'service plan', 'phone number', 'email', 'country', 'account status', 'account details', 'account type',
    'customer since', 'billing/finance', 'current balance', 'revenue till date', 'invoice no', 'invoice date',
    'service details', 'device info', 'profiles', 'primary profile', 'payment methods', 'payment method type',
    'details', 'threshold', 'max. devices per year',
  ]);
  const EMPTY_VALUES = new Set(['', '-', '--', '---', 'n/a', 'na', 'null', 'none', 'undefined', 'no data available']);
  const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
  const STATUS_WORD_RE = /^(active|inactive|suspended|cancell?ed|closed|pending|terminated|expired|deactivated|blocked|trial|paused|churned)$/i;

  const isEmptyValue = (v) => EMPTY_VALUES.has(String(v == null ? '' : v).trim().toLowerCase());

  function normalizeText(raw) {
    return String(raw || '')
      .replace(/\r\n?/g, '\n')
      .replace(/[\u00a0\u2007\u202f\u2009]/g, ' ')
      .replace(/[\u200b\u200e\u200f\u202a-\u202e\u2066-\u2069\ufeff]/g, '')   // invisible direction marks
      .replace(/[\u0660-\u0669]/g, (d) => String(d.charCodeAt(0) - 0x660))     // Arabic digits -> 0-9
      .replace(/[\u06f0-\u06f9]/g, (d) => String(d.charCodeAt(0) - 0x6f0));
  }

  /** "+966 59-999 2401" / "00966599992401" -> "966599992401". Returns null if not a phone. */
  function normalizePhone(v) {
    if (v == null || isEmptyValue(v)) return null;
    let s = normalizeText(v).replace(/[^\d+]/g, '');
    s = s.replace(/^\+/, '').replace(/\+/g, '');
    if (s.startsWith('00')) s = s.slice(2);
    return /^\d{6,15}$/.test(s) ? s : null;
  }

  /** Same number written two ways? ("966599992401" and "0599992401") */
  function samePhone(a, b) {
    if (a === b) return true;
    const x = a.replace(/^0+/, '');
    const y = b.replace(/^0+/, '');
    const [short, long] = x.length <= y.length ? [x, y] : [y, x];
    return short.length >= 8 && long.endsWith(short);
  }

  function removeCaseInsensitive(text, part) {
    if (!part) return text;
    const i = text.toLowerCase().indexOf(part.toLowerCase());
    return i < 0 ? text : text.slice(0, i) + ' ' + text.slice(i + part.length);
  }

  function parseEvergent(raw) {
    const text = normalizeText(raw);
    const lines = text.split('\n').map((s) => s.replace(/\s+/g, ' ').trim()).filter(Boolean);
    const lower = lines.map((s) => s.toLowerCase());
    const r = {
      cpid: null, spid: null, lastName: null, nameNote: null, email: null,
      phone: null, whatsapp: null, phoneNote: null, whatsappNote: null, country: null, countryRaw: null,
      countryCode: null, countryCodeNote: null, status: null, warnings: [], errors: [],
    };

    /** All values written on the line after a label ("CP Customer ID" -> next line). */
    const valuesAfter = (label, from = 0) => {
      const out = [];
      for (let i = from; i < lines.length - 1; i++) {
        if (lower[i] === label && !LABELS.has(lower[i + 1])) out.push(lines[i + 1]);
      }
      return out;
    };
    const firstValue = (label) => valuesAfter(label).find((v) => !isEmptyValue(v)) || null;
    const profileIdx = lower.indexOf('primary profile');

    // --- IDs (labels first, Evergent links as backup) ---
    const cpids = valuesAfter('cp customer id').filter((v) => /^\d{6,25}$/.test(v));
    const linkPairs = [...text.matchAll(/accountDetails\/(?:[A-Za-z]+\/)+(\d{6,25})\/([A-Za-z0-9-]{8,64})/g)]
      .map((m) => ({ cpid: m[1], spid: m[2] }));
    r.cpid = cpids[0] || (linkPairs[0] && linkPairs[0].cpid) || null;
    const spids = valuesAfter('sp account id').filter((v) => /^[A-Za-z0-9-]{8,64}$/.test(v));
    const linkSp = linkPairs.find((p) => p.cpid === r.cpid);
    r.spid = spids[0] || (linkSp && linkSp.spid) || null;
    const distinct = new Set([...cpids, ...linkPairs.map((p) => p.cpid)]);
    if (distinct.size > 1) r.warnings.push(`The paste has ${distinct.size} different accounts. Used CPID ${r.cpid}.`);

    r.status = firstValue('account status');

    // --- Email ---
    const emailValue = firstValue('email');
    if (emailValue && EMAIL_RE.test(emailValue)) r.email = emailValue;

    const summaryIdx = lower.indexOf('account summary');
    let summaryLine = '';
    if (summaryIdx >= 0 && summaryIdx + 1 < lines.length && !LABELS.has(lower[summaryIdx + 1])) {
      summaryLine = lines[summaryIdx + 1].replace(/^[*•\-–]\s*/, '');
    }
    if (!r.email) {
      const tokens = [...summaryLine.split(' ')];
      if (profileIdx >= 0) for (let i = profileIdx + 1; i < Math.min(lines.length, profileIdx + 4); i++) tokens.push(lines[i]);
      r.email = tokens.find((t) => EMAIL_RE.test(t)) || null;
    }

    // --- Phones ---
    const phoneNumber = normalizePhone(firstValue('phone number'));
    let cellPhone = null;
    const cellRe = /^cell\s*phone\s*:?\s*/i;
    let cellIdx = lines.findIndex((l, i) => i > profileIdx && cellRe.test(l));
    if (cellIdx < 0) cellIdx = lines.findIndex((l) => cellRe.test(l));
    if (cellIdx >= 0) {
      let rest = lines[cellIdx].replace(cellRe, '');
      if (!rest && cellIdx + 1 < lines.length) rest = lines[cellIdx + 1];
      rest = removeCaseInsensitive(rest, r.email);           // Evergent glues the email to the number
      rest = rest.replace(/--+\s*$/, '').trim();
      const lead = rest.match(/^\+?[\d\s\-().]+/);
      const digits = lead ? lead[0].replace(/\D/g, '') : '';
      if (phoneNumber && digits.length > phoneNumber.length && digits.startsWith(phoneNumber)) {
        cellPhone = phoneNumber;                              // email digits stuck to the number
      } else if (digits.length > 15) {
        r.warnings.push('The "Cell Phone" number is not clear in the paste. It was not used.');
      } else {
        cellPhone = normalizePhone(lead ? lead[0] : '');
      }
    }
    if (phoneNumber && cellPhone && !samePhone(phoneNumber, cellPhone)) {
      r.phone = phoneNumber;
      r.whatsapp = cellPhone;
      r.phoneNote = 'Phone Number and Cell Phone are different';
    } else {
      r.phone = phoneNumber || cellPhone || null;
      if (!phoneNumber && cellPhone) r.phoneNote = 'taken from Cell Phone';
      if (phoneNumber && cellPhone) r.whatsappNote = 'same as Phone, not changed';
    }

    // --- Name -> Last Name ---
    const cleanName = (s) => {
      let n = String(s || '');
      if (r.email) n = removeCaseInsensitive(n, r.email);
      n = n.replace(/\S+@\S+\.\S+/g, ' ');
      n = n.replace(/(^|\s)--+(?=\s|$)/g, ' ');
      n = n.replace(/\s+[-–—]+\s+/g, ' ').replace(/^\s*[-–—*•]+\s*/, '').replace(/\s*[-–—]+\s*$/, '');
      return n.replace(/\s+/g, ' ').trim();
    };
    const fromSummary = () => {
      let s = summaryLine;
      if (r.email) s = removeCaseInsensitive(s, r.email);
      s = s.replace(/(^|\s)--+(?=\s|$)/g, ' ').replace(/\s+/g, ' ').trim();
      const tokens = s.split(' ');
      const last = tokens[tokens.length - 1] || '';
      if ((r.status && last.toLowerCase() === r.status.toLowerCase()) || STATUS_WORD_RE.test(last)) tokens.pop();
      return cleanName(tokens.join(' '));
    };
    const fromProfile = () => {
      if (profileIdx < 0 || profileIdx + 1 >= lines.length) return '';
      const cand = lines[profileIdx + 1];
      if (cellRe.test(cand) || EMAIL_RE.test(cand) || LABELS.has(lower[profileIdx + 1])) return '';
      return cleanName(cand);
    };
    const nameProblem = (n) => {
      if (!n) return 'no name in Evergent';
      if (/[?\uFFFD]/.test(n)) return `name is not readable ("${n}")`;
      if (n.includes('@')) return 'name is an email';
      const words = n.toLowerCase().replace(/[^\p{L}\s]/gu, ' ').replace(/\s+/g, ' ').trim();
      if (CONFIG.IGNORED_NAMES.includes(words)) return `name is "${n}"`;
      const letters = (n.match(/\p{L}/gu) || []).length;
      if (letters < CONFIG.MIN_NAME_LETTERS) return `"${n}" has less than ${CONFIG.MIN_NAME_LETTERS} letters`;
      return null;
    };
    const candidates = [fromProfile(), fromSummary()];
    const good = candidates.find((n) => n && !nameProblem(n));
    if (good) r.lastName = good.slice(0, 80);
    else r.nameNote = 'ignored: ' + nameProblem(candidates.find(Boolean) || '');

    // --- Country + Country Code ---
    r.countryRaw = firstValue('country');
    if (r.countryRaw) {
      r.country = resolveCountry(r.countryRaw);
      if (!r.country) r.warnings.push(`Unknown country "${r.countryRaw}". Choose Country and Country Code by hand.`);
    }
    if (r.country) {
      r.countryCode = r.country.dial;
    } else if (r.phone) {
      r.countryCode = dialFromPhone(r.phone);
      if (r.countryCode) r.countryCodeNote = 'guessed from the phone number';
    }

    // --- Sanity check ---
    if (!r.cpid && !r.spid && !r.email && !r.phone && !r.country) {
      r.errors.push(/no account found/i.test(text)
        ? 'Evergent says "No account found". Nothing to fill.'
        : 'This does not look like an Evergent account page. Copy the page with Ctrl+A, Ctrl+C.');
    }
    return r;
  }

  /* =====================================================================
   * 4. DOM HELPERS (work with Salesforce Shadow DOM, synthetic or native)
   * ===================================================================== */
  const sleep = (ms) => new Promise((res) => setTimeout(res, ms));
  const attrEq = (v) => '"' + String(v).replace(/["\\]/g, '\\$&') + '"';

  /** document + every shadow root under `start`. */
  function collectRoots(start = document) {
    const roots = [];
    const seen = new Set();
    const stack = [start];
    if (start && start.shadowRoot) stack.push(start.shadowRoot);
    while (stack.length) {
      const root = stack.pop();
      if (!root || seen.has(root) || !root.querySelectorAll) continue;
      seen.add(root);
      roots.push(root);
      const all = root.querySelectorAll('*');
      for (let i = 0; i < all.length; i++) {
        const sr = all[i].shadowRoot;
        if (sr && !seen.has(sr)) stack.push(sr);
      }
    }
    return roots;
  }

  /** querySelectorAll that also looks inside shadow roots. Use simple selectors only. */
  function deepQueryAll(selector, start = document) {
    const out = new Set();
    for (const root of collectRoots(start)) {
      const found = root.querySelectorAll(selector);
      for (let i = 0; i < found.length; i++) out.add(found[i]);
    }
    return [...out];
  }

  function parentAcross(n) {
    if (!n) return null;
    if (n.parentElement) return n.parentElement;
    if (n.parentNode && n.parentNode.host) return n.parentNode.host;
    const root = n.getRootNode ? n.getRootNode() : null;
    return root && root !== n && root.host ? root.host : null;
  }

  function closestDeep(el, selector) {
    for (let n = el; n; n = parentAcross(n)) {
      if (n.nodeType === 1 && n.matches(selector)) return n;
    }
    return null;
  }

  function isVisible(el) {
    if (!el || !el.isConnected || !el.getClientRects().length) return false;
    return getComputedStyle(el).visibility !== 'hidden';
  }

  const DIALOG_SEL = 'section[role="dialog"], [role="dialog"], .slds-modal, .uiModal';
  /** Several matches (cached hidden tabs, modal over a page)? Take the visible one, a modal first. */
  function pickBest(els) {
    const visible = els.filter(isVisible);
    return visible.find((e) => closestDeep(e, DIALOG_SEL)) || visible[0] || null;
  }

  /**
   * Text of an element as the user sees it, without screen-reader-only text.
   * Salesforce can hide component parts from normal DOM reading, so the text is read
   * two ways (through shadow roots, and as the browser draws the page) and the
   * fuller result wins.
   */
  function deepText(node) {
    const a = composedText(node);
    const b = physicalText(node);
    return b.length > a.length ? b : a;
  }

  const SKIP_TAGS = new Set(['style', 'script', 'template', 'svg', 'button']);

  /** Text as the browser draws it. Works even when Salesforce hides component parts. */
  function physicalText(node) {
    const parts = [];
    let walker;
    try {
      walker = document.createTreeWalker(node, NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT, {
        acceptNode(n) {
          if (n.nodeType === 3) return NodeFilter.FILTER_ACCEPT;
          if (n !== node && (SKIP_TAGS.has(n.localName) || n.classList.contains('slds-assistive-text'))) return NodeFilter.FILTER_REJECT;
          return NodeFilter.FILTER_SKIP;
        },
      });
    } catch (e) { return ''; }
    while (walker.nextNode()) parts.push(walker.currentNode.nodeValue);
    return parts.join(' ').replace(/\s+/g, ' ').trim();
  }

  /** Every element once: the page as drawn, plus open shadow roots. */
  function walkElements(visit, start) {
    const stack = [start || document.body || document.documentElement];
    const seen = new Set();
    while (stack.length) {
      const root = stack.pop();
      if (!root || seen.has(root)) continue;
      seen.add(root);
      let walker;
      try { walker = document.createTreeWalker(root, NodeFilter.SHOW_ELEMENT); } catch (e) { continue; }
      for (let el = walker.currentNode; el; el = walker.nextNode()) {
        if (el.nodeType !== 1) continue;
        visit(el);
        const sr = el.shadowRoot;
        if (sr && !seen.has(sr)) stack.push(sr);
      }
    }
  }

  /** Text through shadow roots and slots. */
  function composedText(node) {
    const parts = [];
    const visit = (n) => {
      if (n.nodeType === 3) { parts.push(n.nodeValue); return; }
      if (n.nodeType !== 1 && n.nodeType !== 11) return;
      if (n.nodeType === 1) {
        const tag = n.localName;
        if (tag === 'style' || tag === 'script' || tag === 'template' || tag === 'svg') return;
        if (n.classList.contains('slds-assistive-text')) return;
        if (tag === 'button' && n !== node) return;      // edit/preview buttons are not part of the value
        if (tag === 'slot') {
          const assigned = typeof n.assignedNodes === 'function' ? n.assignedNodes() : [];
          (assigned.length ? assigned : [...n.childNodes]).forEach(visit);
          return;
        }
        if (n.shadowRoot) { [...n.shadowRoot.childNodes].forEach(visit); return; }
      }
      [...n.childNodes].forEach(visit);
    };
    visit(node);
    return parts.join(' ').replace(/\s+/g, ' ').trim();
  }

  async function waitFor(fn, timeout = 4000, every = 100) {
    const end = Date.now() + timeout;
    for (;;) {
      let v = null;
      try { v = fn(); } catch (e) { v = null; }
      if (v) return v;
      if (Date.now() > end) return null;
      await sleep(every);
    }
  }

  function fireClick(el) {
    const opts = { bubbles: true, composed: true, cancelable: true, view: window, button: 0 };
    const P = window.PointerEvent || MouseEvent;
    el.dispatchEvent(new P('pointerdown', opts));
    el.dispatchEvent(new MouseEvent('mousedown', opts));
    el.dispatchEvent(new P('pointerup', opts));
    el.dispatchEvent(new MouseEvent('mouseup', opts));
    el.dispatchEvent(new MouseEvent('click', opts));
  }

  function routeInfo() {
    const p = location.pathname;
    const rec = p.match(/^\/lightning\/r\/(?:([A-Za-z0-9_]+)\/)?([A-Za-z0-9]{15,18})\/(view|edit)/);
    if (rec) return { object: rec[1] || (rec[2].startsWith('001') ? 'Account' : null), recordId: rec[2], mode: rec[3] };
    const nw = p.match(/^\/lightning\/o\/([A-Za-z0-9_]+)\/new/);
    if (nw) return { object: nw[1], recordId: null, mode: 'new' };
    return { object: null, recordId: null, mode: null };
  }

      if (btn) return btn;

  function setNativeValue(input, value) {
    const proto = input instanceof HTMLTextAreaElement ? HTMLTextAreaElement.prototype : HTMLInputElement.prototype;
    const setter = Object.getOwnPropertyDescriptor(proto, 'value').set;
    input.focus({ preventScroll: true });
    setter.call(input, value);
    input.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
    input.dispatchEvent(new Event('change', { bubbles: true, composed: true }));
    input.blur();
  }

  async function fillText(field, value) {
    const input = findTextInput(field);
    if (!input) return { status: 'error', msg: 'field not found on this page' };
    let v = String(value);
    let msg = '';
    if (input.maxLength > 0 && v.length > input.maxLength) { v = v.slice(0, input.maxLength); msg = 'cut to ' + input.maxLength + ' characters'; }
    if (input.value === v) return { status: 'same', el: input };
    setNativeValue(input, v);
    await sleep(60);
    const now = findTextInput(field) || input;
    if (now.value !== v) return { status: 'error', msg: 'Salesforce did not keep the value', el: input };
    return { status: 'ok', el: input, msg };
  }

  /** Options of an open picklist (they live next to the button, or in a popup). */
  function findOptions(btn) {
    const sel = 'lightning-base-combobox-item, [role="option"]';
    const valid = (o) => isVisible(o) && !o.closest('.sfx-ui');
    let opts = deepQueryAll(sel, btn.getRootNode()).filter(valid);
    if (!opts.length) {
      const listId = btn.getAttribute('aria-controls');
      opts = deepQueryAll(sel).filter(valid).filter((o) => {
        const lb = closestDeep(o, '[role="listbox"]');
        return !listId || !lb || lb.id === listId;
      });
    }
    return opts;
  }

  function optionInfo(o) {
    const value = o.getAttribute('data-value') || '';
    const label = deepText(o) || o.getAttribute('aria-label') || o.getAttribute('title') || value;
    return { value, label };
  }

  function closeDropdown(btn) {
    if (btn && btn.isConnected && btn.getAttribute('aria-expanded') === 'true') {
      btn.dispatchEvent(new KeyboardEvent('keydown', { key: 'Escape', code: 'Escape', bubbles: true, composed: true }));
      if (btn.getAttribute('aria-expanded') === 'true') btn.click();
    }
  }

  async function fillPicklist(field, wanted) {
    const btn = await waitFor(() => findCombobox(field), 1500);
    if (!btn) return { status: 'error', msg: 'field not found on this page' };
    const unlocked = await waitFor(() => btn.getAttribute('aria-disabled') !== 'true' && !btn.disabled, 2500);
    if (!unlocked) return { status: 'error', msg: 'field is locked', el: btn };

    const current = { value: btn.getAttribute('data-value') || '', label: deepText(btn) };
    if (wanted.score(current.value, current.label) > 0) return { status: 'same', el: btn };

    if (btn.getAttribute('aria-expanded') !== 'true') { btn.focus({ preventScroll: true }); btn.click(); }
    const opts = await waitFor(() => { const o = findOptions(btn); return o.length ? o : null; }, 4000);
    if (!opts) { closeDropdown(btn); return { status: 'error', msg: 'the list did not open', el: btn }; }

    let best = null;
    let bestScore = 0;
    for (const o of opts) {
      const info = optionInfo(o);
      if (!info.value || /^-+\s*none\s*-+$/i.test(info.value)) continue;
      const s = wanted.score(info.value, info.label);
      if (s > bestScore) { best = o; bestScore = s; }
    }
    if (!best) { closeDropdown(btn); return { status: 'error', msg: `"${wanted.display}" is not in the list. Choose it by hand.`, el: btn }; }

    const target = optionInfo(best).value;
    if (best.scrollIntoView) best.scrollIntoView({ block: 'nearest' });
    fireClick(best);
    const done = await waitFor(() => {
      const b = btn.isConnected ? btn : findCombobox(field);
      return b && b.getAttribute('data-value') === target;
    }, 2500);
    if (!done) { closeDropdown(btn); return { status: 'error', msg: 'could not select it. Choose it by hand.', el: btn }; }
    return { status: 'ok', el: btn };
  }

  function countryWanted(c) {
    const names = new Set([c.name, ...c.aliases].map(normName));
    return {
      display: c.name,
      score: (value, label) => {
        if (value && value.toUpperCase() === c.iso) return 3;          // picklist stores ISO codes
        if (names.has(normName(value)) || names.has(normName(label))) return 2;
        return 0;
      },
    };
  }

  function dialWanted(dial, c) {
    const names = c ? new Set([c.name, ...c.aliases].map(normName)) : new Set();
    const digits = (s) => String(s || '').replace(/\D/g, '');
    return {
      display: dial,
      score: (value, label) => {
        const dv = digits(value);
        const dl = digits(label);
        if (dv !== dial && !(dv === '' && dl === dial)) return dl === dial ? 1 : 0;
        // Shared codes (+1, +7, +44...): prefer the option that names this country.
        return names.has(normName(label).replace(/\d+/g, '')) ? 3 : 2;
      },
    };
  }

  let busy = false;
  async function fillSalesforce(data) {
    if (busy) return { error: 'Already filling, please wait.' };
    busy = true;
    const results = {};
    try {
      await ensureEditMode();
      const values = {
        cpid: data.cpid, spid: data.spid, lastName: data.lastName, email: data.email,
        phone: data.phone, whatsapp: data.whatsapp,
      };
      for (const key of FILL_ORDER) {
        const field = FIELDS[key];
        let res;
        if (field.kind === 'text') {
          res = values[key] ? await fillText(field, values[key]) : { status: 'skip' };
        } else if (key === 'country') {
          res = data.country ? await fillPicklist(field, countryWanted(data.country)) : { status: 'skip' };
        } else {
          res = data.countryCode ? await fillPicklist(field, dialWanted(data.countryCode, data.country)) : { status: 'skip' };
        }
        results[key] = res;
        if (res.el && (res.status === 'ok' || res.status === 'error')) flash(res.el, res.status === 'ok');
        if (key === 'country' && res.status === 'ok') await sleep(400);   // Country Code may depend on Country
        else await sleep(80);
      }
      return { results };
    } catch (e) {
      return { results, error: e.message || String(e) };
    } finally {
      busy = false;
    }
  }

  function h(tag, attrs, ...children) {
    const el = document.createElement(tag);
    for (const [k, v] of Object.entries(attrs || {})) {
      if (v == null || v === false) continue;
      if (k === 'class') el.className = v;
      else if (k === 'text') el.textContent = v;
      else if (k.startsWith('on')) el.addEventListener(k.slice(2), v);
      else if (k in el && typeof v !== 'string') el[k] = v;
      else el.setAttribute(k, v === true ? '' : v);
    }
    for (const c of children.flat()) if (c != null) el.append(c instanceof Node ? c : document.createTextNode(String(c)));
    return el;
  }

    ui.toast = h('div', { class: 'sfx-ui sfx-toast', role: 'status' });
    document.body.append(ui.fab, ui.panel, ui.toast);
  }

  function togglePanel(force) {
    const show = typeof force === 'boolean' ? force : ui.panel.hidden;
    if (show) checkRoute();
    ui.panel.hidden = !show;
    if (show) setTimeout(() => ui.ta.focus(), 0);
  }

  /** New profile / other profile opened: forget the previous paste. */
  let lastRouteKey = null;
  function checkRoute() {
    const key = routeKey();
    if (key === lastRouteKey) return;
    const first = lastRouteKey === null;
    lastRouteKey = key;
    if (!first && !busy) { ui.ta.value = ''; clearAll(); }
  }

  let toastTimer = null;
  function toast(text, ms = 1800) {
    ui.toast.textContent = text;
    ui.toast.classList.add('sfx-show');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => ui.toast.classList.remove('sfx-show'), ms);
  }

  function setMessage(text, kind = 'info') {
    ui.msg.replaceChildren();
    if (text) ui.msg.append(h('div', { class: 'sfx-msg sfx-' + kind, text }));
  }
  function addMessage(text, kind = 'info') { ui.msg.append(h('div', { class: 'sfx-msg sfx-' + kind, text })); }

  function clearAll() { ui.table.replaceChildren(); setMessage(''); }

  const RESULT_TEXT = {
    ok: ['✓ filled', 'sfx-ok'], same: ['✓ already correct', 'sfx-ok'], skip: ['— not in Evergent', 'sfx-dim'], error: ['✗ ', 'sfx-err'],
  };

  function renderTable(d, results) {
    const rows = [
      ['cpid', 'CPID', d.cpid],
      ['spid', 'SP/Gigya ID', d.spid],
      ['lastName', 'Last Name', d.lastName, d.nameNote],
      ['email', 'Email', d.email],
      ['phone', 'Phone', d.phone, d.phoneNote],
      ['whatsapp', 'WhatsApp', d.whatsapp, d.whatsappNote],
      ['country', 'Country', d.country ? `${d.country.name} (${d.country.iso})` : d.countryRaw, d.country ? null : (d.countryRaw ? 'unknown country code' : null)],
      ['countryCode', 'Country Code', d.countryCode, d.countryCodeNote],
    ];
    ui.table.replaceChildren(...rows.map(([key, label, value, note]) => {
      const res = results && results[key];
      const noteCls = /^(ignored|unknown)/.test(note || '') ? 'sfx-warn' : 'sfx-dim';
      let text = note || (value ? 'ready' : '—');
      let cls = noteCls;
      if (res) {
        const [t, c] = RESULT_TEXT[res.status] || ['', ''];
        text = res.status === 'error' ? t + res.msg : (res.status === 'skip' && note ? note : t + (res.msg ? ' (' + res.msg + ')' : ''));
        cls = res.status === 'skip' && note ? noteCls : c;
      }
      return h('tr', null,
        h('td', { text: label }),
        h('td', { class: 'sfx-val', dir: 'auto', text: value || '—' }),
        h('td', { class: 'sfx-res ' + cls, text }));
    }));
  }

  function showParsed(d) {
    renderTable(d);
    setMessage('');
    d.errors.forEach((m) => addMessage(m, 'bad'));
    d.warnings.forEach((m) => addMessage(m, 'mid'));
  }

  /** fromPage = Ctrl+V on the page: the panel stays closed unless something needs attention. */
  async function runFill(text, fromPage = false) {
    if (!String(text || '').trim()) { setMessage('Paste the Evergent page first.', 'mid'); return; }
    if (busy) { toast('Still filling, please wait…'); return; }
    const d = parseEvergent(text);
    showParsed(d);
    if (d.errors.length) { if (fromPage) toast(d.errors[0], 3500); return; }
    if (!onAccountForm()) { addMessage('Open an Account page (or New Account) to fill it.', 'mid'); return; }
    [ui.btnClip, ui.btnFill].forEach((b) => { b.disabled = true; });
    addMessage('Filling…', 'info');
    if (fromPage) toast('Filling from Evergent…', 10000);
    const out = await fillSalesforce(d);
    [ui.btnClip, ui.btnFill].forEach((b) => { b.disabled = false; });
    renderTable(d, out.results);
    setMessage('');
    if (out.error) addMessage(out.error, 'bad');
    d.warnings.forEach((m) => addMessage(m, 'mid'));
    const failed = Object.values(out.results || {}).filter((r) => r.status === 'error').length;
    if (!out.error) {
      addMessage(failed ? `Done with ${failed} problem(s) (red). Fix them, check the values, then click Save.`
        : 'Done. Check the values, then click Save.', failed ? 'mid' : 'info');
      toast(failed ? 'Filled with problems — see the panel' : 'Filled — check and click Save', 2500);
    } else {
      toast(out.error, 3500);
    }
    if (fromPage && (failed || out.error)) togglePanel(true);
  }

  /** Clipboard HTML may hold the Evergent links even when the plain text does not. */
  function linksFromHtml(html) {
    if (!html) return '';
    return [...html.matchAll(/href=["']([^"']*accountDetails[^"']*)["']/gi)].map((m) => m[1]).join('\n');
  }

  function onPaste(e) {
    const cd = e.clipboardData;
    if (!cd) { setTimeout(() => afterPaste(ui.ta.value), 0); return; }
    e.preventDefault();
    const text = cd.getData('text/plain') || '';
    const links = linksFromHtml(cd.getData('text/html'));
    ui.ta.value = links ? text + '\n' + links : text;
    afterPaste(ui.ta.value);
  }

  function afterPaste(text) {
    if (ui.auto.checked) { runFill(text); return; }
    const d = parseEvergent(text);
    showParsed(d);
    if (!d.errors.length) addMessage('Click "Fill" to put these values in the form.', 'info');
  }

  async function pasteFromClipboard() {
    togglePanel(true);
    let text = '';
    try {
      text = await navigator.clipboard.readText();
    } catch (e) {
      setMessage('The browser blocked clipboard reading. Click in the box and press Ctrl+V.', 'mid');
      ui.ta.focus();
      return;
    }
    ui.ta.value = text;
    runFill(text);
  }

  /* ---------- Ctrl+V anywhere on the page ---------- */
  const NON_TEXT_INPUT = /^(button|checkbox|radio|submit|reset|image|file|color|range|hidden)$/i;
  function isEditable(el) {
    if (!el || el.nodeType !== 1) return false;
    if (el.isContentEditable) return true;
    if (el.localName === 'textarea') return !el.readOnly && !el.disabled;
    if (el.localName === 'input') return !NON_TEXT_INPUT.test(el.type || 'text') && !el.readOnly && !el.disabled;
    const role = el.getAttribute('role');
    return role === 'textbox' || role === 'searchbox';
  }

  function deepActiveElement() {
    let a = document.activeElement;
    while (a && a.shadowRoot && a.shadowRoot.activeElement) a = a.shadowRoot.activeElement;
    return a;
  }

  /** A full Evergent page (never something you want inside one field). */
  function isFullEvergentPage(text) {
    const lines = normalizeText(text).split('\n').map((l) => l.trim().toLowerCase());
    return lines.includes('cp customer id') && lines.includes('sp account id');
  }

  function onDocPaste(e) {
    if (!onAccountForm()) return;
    const path = e.composedPath ? e.composedPath() : [];
    if (path.some((n) => n && n.nodeType === 1 && n.classList.contains('sfx-ui'))) return;   // our own box
    const cd = e.clipboardData;
    const plain = cd ? cd.getData('text/plain') || '' : '';
    if (!plain.trim()) return;                                        // image or nothing: not ours

    const inField = [path[0], e.target, deepActiveElement()].some(isEditable);
    if (inField && !(CONFIG.CATCH_EVERGENT_IN_INPUTS && isFullEvergentPage(plain))) return;   // normal paste

    checkRoute();
    const links = linksFromHtml(cd.getData('text/html'));
    const text = links ? plain + '\n' + links : plain;
    if (parseEvergent(text).errors.length) {
      if (!inField) toast('Not an Evergent account page — nothing filled', 2500);
      return;
    }
    e.preventDefault();
    e.stopPropagation();
    if (inField && document.activeElement && document.activeElement.blur) document.activeElement.blur();
    ui.ta.value = text;
    if (ui.auto.checked) runFill(text, true);
    else { togglePanel(true); afterPaste(text); }
  }

  /* =====================================================================
   * 7. ONE-CLICK COPY (double click still opens edit mode)
   * ===================================================================== */
  let copyTimer = null;

  async function copyText(text) {
    try {
      await navigator.clipboard.writeText(text);
      return true;
    } catch (e) {
      const active = document.activeElement;
      const ta = h('textarea', { class: 'sfx-ui' });
      ta.value = text;
      ta.style.cssText = 'position:fixed;top:-1000px;left:-1000px;opacity:0;';
      document.body.appendChild(ta);
      ta.select();
      let ok = false;
      try { ok = document.execCommand('copy'); } catch (err) { ok = false; }
      ta.remove();
      if (active && active.focus) active.focus({ preventScroll: true });
      return ok;
    }
  }

  const pointIn = (el, x, y) => {
    const r = el.getBoundingClientRect();
    return r.width > 0 && x >= r.left && x <= r.right && y >= r.top && y <= r.bottom;
  };

  /** Backup when the browser hides the inner click target: find the field under the mouse. */
  function valueAtPoint(x, y) {
    const sel = 'span.test-id__field-value.slds-form-element__static';
    let hit = deepQueryAll(sel).find((s) => pointIn(s, x, y)) || null;
    if (!hit) walkElements((el) => { if (!hit && el.matches(sel) && pointIn(el, x, y)) hit = el; });
    return hit;
  }

  function onDocClick(e) {
    if (e.button !== 0 || e.ctrlKey || e.metaKey || e.shiftKey || e.altKey) return;
    if (!onRecordOf(CONFIG.COPY_ON)) return;
    let valueEl = null;
    let link = null;
    let insideLayout = false;
    for (const n of e.composedPath()) {
      if (!n || n.nodeType !== 1) continue;
      if (n.classList.contains('sfx-ui')) return;
      if (n.matches('button, input, textarea, select, [role="combobox"], [contenteditable="true"]')) return;
      if (!link && n.localName === 'a') link = n;
      if (n.classList.contains('test-id__field-value')) { valueEl = n; break; }
      if (n.localName === 'records-record-layout-item' || n.localName === 'records-lwc-detail-panel') { insideLayout = true; break; }
    }
    if (!valueEl && !insideLayout) {
      valueEl = valueAtPoint(e.clientX, e.clientY);
      if (valueEl) link = deepQueryAll('a', valueEl).find((a) => pointIn(a, e.clientX, e.clientY)) || null;
    }
    if (!valueEl || !valueEl.classList.contains('slds-form-element__static')) return;
    const root = closestDeep(valueEl, '.test-id__output-root');
    if (root && root.classList.contains('recordlayout_in_editing_mode')) return;

    if (link) { e.preventDefault(); e.stopPropagation(); }   // copy instead of calling / opening email
    clearTimeout(copyTimer);
    if (e.detail > 1) return;                               // 2nd click of a double click
    if (String(window.getSelection ? window.getSelection() : '').trim()) return;   // user is selecting text

    copyTimer = setTimeout(async () => {
      const text = deepText(valueEl);
      if (!text) { toast('This field is empty'); return; }
      const ok = await copyText(text);
      if (ok) { flash(valueEl, true); toast('Copied: ' + text); } else toast('Could not copy');
    }, CONFIG.COPY_DELAY_MS);
  }
  function onDocDblClick() { clearTimeout(copyTimer); }

  /* =====================================================================
   * 8. STATUS COLORS
   * ===================================================================== */
  const normStatus = (s) => String(s || '').replace(/\s+/g, ' ').trim().toLowerCase();
  const STATUS_MAP = new Map(Object.entries(CONFIG.STATUS_STYLES).map(([k, v]) => [normStatus(k), v]));
  const onHighlightPage = () => CONFIG.HIGHLIGHT_PAGES.some((re) => re.test(location.pathname));
  const CELL_SEL = 'td, th, [role="gridcell"], [role="rowheader"]';
  const isStatusColumn = (cell) => CONFIG.STATUS_COLUMN_RE.test(cell.getAttribute('data-label') || '');

  /** Table cell that contains `el` (crosses component borders). */
  function cellOf(el) {
    for (let n = el, i = 0; n && i < 50; i++) {
      const c = n.closest ? n.closest(CELL_SEL) : null;
      if (c) return c;
      const root = n.getRootNode ? n.getRootNode() : null;
      n = root && root.host ? root.host : null;
    }
    return null;
  }

  /**
   * Status cells on the page -> status read from a title attribute ('' = read the text later).
   * A cell counts if its column is "Status", or if it has no column name but holds a status title.
   */
  function findStatusCells() {
    const cells = new Map();
    const fromTitle = (el, title) => {
      const cell = cellOf(el);
      if (cell && (!cell.hasAttribute('data-label') || isStatusColumn(cell)) && !cells.get(cell)) cells.set(cell, title);
    };
    walkElements((el) => {
      if (el.hasAttribute('data-label') && el.matches(CELL_SEL) && isStatusColumn(el) && !cells.has(el)) cells.set(el, '');
      const t = el.getAttribute('title');
      if (t && STATUS_MAP.has(normStatus(t)) && !el.closest('.sfx-ui')) fromTitle(el, t.trim());
    });
    if (!cells.size) {
      for (const c of deepQueryAll('td[data-label], th[data-label], [role="gridcell"][data-label]')) if (isStatusColumn(c)) cells.set(c, '');
      for (const t of deepQueryAll('[title]')) if (STATUS_MAP.has(normStatus(t.getAttribute('title')))) fromTitle(t, t.getAttribute('title').trim());
    }
    return cells;
  }

  function readStatus(cell, fromTitle) {
    if (fromTitle) return fromTitle;
    for (const t of deepQueryAll('[title]', cell)) {
      const v = (t.getAttribute('title') || '').trim();
      if (STATUS_MAP.has(normStatus(v))) return v;
    }
    return deepText(cell);
  }

  const painted = new Set();
  function unpaint(cell) {
    cell.style.removeProperty('background-color');
    cell.style.removeProperty('border');
    delete cell.dataset.sfxStatus;
    painted.delete(cell);
  }

  function paint(cell, status) {
    const style = STATUS_MAP.get(normStatus(status));
    if (!style) { if (cell.dataset.sfxStatus !== undefined) unpaint(cell); return; }
    // Salesforce redraws cells and can drop our style, so look at the real style every time.
    const missing = !cell.style.getPropertyValue('background-color') || !cell.style.getPropertyValue('border-top-style');
    if (missing || cell.dataset.sfxStatus !== status) {
      cell.style.setProperty('background-color', style.bg, 'important');
      cell.style.setProperty('border', style.border, 'important');
      cell.dataset.sfxStatus = status;
    }
    painted.add(cell);
  }

  function applyHighlights() {
    if (!onHighlightPage()) return;
    const found = findStatusCells();
    for (const [cell, title] of found) paint(cell, readStatus(cell, title));
    for (const cell of [...painted]) {
      if (!cell.isConnected) painted.delete(cell);
      else if (!found.has(cell)) unpaint(cell);   // no longer a status cell
    }
  }

  /** SFX.check() in the console: shows what the color code sees on this page. */
  function statusCheck() {
    const labels = new Map();
    walkElements((el) => {
      const l = el.getAttribute('data-label');
      if (l && el.matches(CELL_SEL)) labels.set(l, (labels.get(l) || 0) + 1);
    });
    const viaSearch = deepQueryAll('td[data-label], th[data-label]').length;
    const found = findStatusCells();
    const rows = [...found].slice(0, 25).map(([cell, title]) => {
      const status = readStatus(cell, title);
      return {
        column: cell.getAttribute('data-label') || '(none)',
        status: status || '(empty)',
        'has a color': STATUS_MAP.has(normStatus(status)) ? 'yes' : 'no',
        'background now': getComputedStyle(cell).backgroundColor,
        visible: isVisible(cell) ? 'yes' : 'no',
      };
    });
    console.group(`[SF Helper v${VERSION}] status colors check`);
    console.log('Page:', location.pathname);
    console.log('Colors allowed on this page:', onHighlightPage() ? 'YES' : 'NO (page is not in HIGHLIGHT_PAGES)');
    console.log('Table cells with a column name:', [...labels.values()].reduce((a, b) => a + b, 0), '| found by normal search:', viaSearch);
    console.log('Column names on this page:', [...labels.keys()].slice(0, 40).join(' | ') || '(none)');
    console.log('Status cells found:', found.size);
    if (rows.length) console.table(rows);
    console.groupEnd();
    return found.size
      ? `Found ${found.size} status cell(s). If they are still not colored, send a screenshot of this to Claude.`
      : 'No status cells found. Send a screenshot of this, and the result of: copy(SFX.statusHtml())';
  }

  /** HTML of the first status row (or of the first status word) for troubleshooting. */
  function statusHtml() {
    const first = [...findStatusCells().keys()][0];
    let el = first ? (first.closest('tr') || first) : null;
    if (!el) {
      walkElements((e) => { if (!el && !e.firstElementChild && STATUS_MAP.has(normStatus(physicalText(e)))) el = e; });
      for (let i = 0; el && i < 8; i++) el = parentAcross(el) || el;
    }
    return el ? el.outerHTML.slice(0, 20000) : 'No status found on this page.';
  }

  /* =====================================================================
   * 9. SCHEDULER (one throttled loop for everything)
   * ===================================================================== */
  let scanTimer = null;
  let lastScan = 0;

  function tick() {
    scanTimer = null;
    if (document.hidden) return;
    lastScan = Date.now();
    checkRoute();
    const onForm = onAccountForm();
    ui.fab.hidden = !onForm;
    if (!onForm && !ui.panel.hidden) togglePanel(false);
    document.documentElement.classList.toggle('sfx-copy-on', onRecordOf(CONFIG.COPY_ON));
    try { applyHighlights(); } catch (e) { console.warn('[SF Helper] highlight error', e); }
  }

  function requestScan() {
    if (scanTimer) return;
    scanTimer = setTimeout(tick, Math.max(0, CONFIG.SCAN_GAP_MS - (Date.now() - lastScan)));
  }

  function onKeyDown(e) {
    if (e.altKey && !e.ctrlKey && !e.metaKey && !e.shiftKey && e.code === CONFIG.HOTKEY_CODE && onAccountForm()) {
      e.preventDefault();
      togglePanel();
    }
  }

  /* =====================================================================
   * 10. START / STOP
   * ===================================================================== */
  buildUI();
  const observer = new MutationObserver(requestScan);
  observer.observe(document.documentElement, { childList: true, subtree: true, characterData: true });
  const heartbeat = setInterval(requestScan, CONFIG.HEARTBEAT_MS);
  const onVisibility = () => { if (!document.hidden) requestScan(); };
  document.addEventListener('click', onDocClick, true);
  document.addEventListener('dblclick', onDocDblClick, true);
  document.addEventListener('keydown', onKeyDown, true);
  document.addEventListener('paste', onDocPaste, true);
  document.addEventListener('visibilitychange', onVisibility);
  tick();

  const api = {
    version: VERSION,
    parse: parseEvergent,
    fill: (text) => runFill(text),
    open: () => togglePanel(true),
    check: statusCheck,
    recolor: applyHighlights,
    statusHtml,
    countries: COUNTRIES,
    destroy() {
      observer.disconnect();
      clearInterval(heartbeat);
      clearTimeout(scanTimer);
      clearTimeout(copyTimer);
      clearTimeout(toastTimer);
      document.removeEventListener('click', onDocClick, true);
      document.removeEventListener('dblclick', onDocDblClick, true);
      document.removeEventListener('keydown', onKeyDown, true);
      document.removeEventListener('paste', onDocPaste, true);
      document.removeEventListener('visibilitychange', onVisibility);
      document.documentElement.classList.remove('sfx-copy-on');
      [ui.fab, ui.panel, ui.toast, ui.style].forEach((n) => n && n.remove());
      if (window[NS] === api) delete window[NS];
      if (window.SFX === api) delete window.SFX;
    },
  };
  window[NS] = api;
  window.SFX = api;
  toast(`SF Helper v${VERSION} is on`, 2500);
  console.info(`[SF Helper] v${VERSION} is on. Ctrl+V on an Account page fills from Evergent. Colors problem? Type: SFX.check()`);
})();
